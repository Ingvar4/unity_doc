## Скрипт Enemy_1.cs

- Пример с горизонтальным перемещением объекта по синусойде из стороны в сторону

```c# title="C# Enemy_1.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

// Замена MonoBehaviour на Enemy критически важна. Именно это делает Enemy_0 подклассом Enemy
// Класс Enemy_1 наследует все не являющиеся закрытыми (non-private) методы, поля, свойства и прочие элементы суперкласса
public class Enemy_1 : Enemy { // Класс Enemy_1 наследует класс Enemy
    [Header("Enemy_1 Inscribed Fields")] 
    [Tooltip("# of seconds for a full sine wave")] // Указывает Unity отображать соответствующую подсказку, когда пользователь наводит курсор мыши на имя поля в окне Inspector.
    public float waveFrequency = 2;
    [Tooltip("Sine wave width in meters")]
    public float waveWidth = 4;
    [Tooltip("Amount the ship will roll left and right with the sine wave")]
    public float waveRotY = 45;

    private float x0; // Начальное значение x в позиции
    private float birthTime;

    void Start() {
        x0 = pos.x; // Установите x0 равным начальной позиции x объекта Enemy_1
        birthTime = Time.time;
    }

    public override void Move() { // Переопределение функции перемещения противника
        // Поскольку pos является свойством, вы не можете напрямую задать pos.x
        // так что получите позицию в виде редактируемого вектора Vector3
        Vector3 tempPos = pos;
        // theta корректируется в зависимости от времени
        float age = Time.time - birthTime;
        float theta = Mathf.PI * 2 * age / waveFrequency;
        float sin = Mathf.Sin(theta);
        tempPos.x = x0 + waveWidth * sin;
        pos = tempPos; // Немного повернуть вокруг оси Y
        Vector3 rot = new Vector3(0, sin * waveRotY, 0);
        this.transform.rotation = Quaternion.Euler(rot);
        // Метод base.Move() по-прежнему обрабатывает перемещение вниз по оси Y
        base.Move(); // Вызов base.Move() обращается к методу Move() суперкласса Enemy. В данном случае метод Move() в подклассе Enemy_1 отвечает 
        // за горизонтальное перемещение по синусоиде, тогда как метод Move() в суперклассе Enemy перемещает игровой объект (GameObject) Enemy_1 вниз по экрану
        // print(bndCheck.isOnScreen);
    }
}
```

## Скрипт Enemy_2.cs

- Пример движения объекта посредством линейной интерполяции с синусоидальным сглаживанием.

```c# title="C# Enemy_2.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Enemy_2 : Enemy {
    [Header("Enemy_2 Inscribed Fields")]
    public float lifeTime = 10;
    [Tooltip("Determines how much the Sine wave will ease the interpolation")]
    public float sinEccentricity = 0.6f; // Изменение этого параметра в инспекторе влияет на скорость движения и более широкую синусойду
    public AnimationCurve rotCurve; // Это добавляет публичное поле типа AnimationCurve, которое можно редактировать в окне Inspector

    [Header("Enemy_2 Private Fields")] // Переменные приватные, однако [SerializeField] позволяет отображать и изменять их в инспекторе, сохраняя при этом защиту от доступа к ним со стороны других классов
    [SerializeField] private float birthTime; // Время начала интерполяции
    private Quaternion baseRotation; // Хранит ориентацию объекта Enemy_2, когда он находится в точке p0 и смотрит (то есть направлен) на точку p1
    [SerializeField] private Vector3 p0, p1; // Lerp_points

    void Start() {
        // Выбор случайной точки у левого края экрана
        p0 = Vector3.zero; // Задаём нулевую точку координат (0, 0, 0)
        p0.x = -bndCheck.camWidth - bndCheck.radius; // -30 - 1f  
        p0.y = Random.Range(-bndCheck.camHeight, bndCheck.camHeight); // -40, 40
        // Выбор случайной точки у правого края экрана
        p1 = Vector3.zero; 
        p1.x = bndCheck.camWidth + bndCheck.radius; // 30 + 1f
        p1.y = Random.Range( -bndCheck.camHeight, bndCheck.camHeight ); // -40, 40
        // Возможная смена сторон
        if (Random.value > 0.5f) {
            p0.x *= -1; // Установка отрицательного значения .x для каждой точки переместит её
            p1.x *= -1; // на другую сторону экрана
        }
        // Привязка появления к текущему времени
        birthTime = Time.time; // Переменная birthTime используется для интерполяции в функции Move()
        // Настройка начального поворота корабля
        transform.position = p0; // Размещает Enemy_2 в точке p0, разворачивает его в сторону p1
        transform.LookAt(p1, Vector3.back); //Вызов transform.LookAt(p1, Vector3.back) направляет ось transform.forward данного объекта на точку p1. 
        // Vector3.back, указывает методу LookAt(), что после поворота на p1 локальная ось Y (ось «вверх») объекта должна быть направлена ​​назад вдоль глобальной оси -Z (то есть в сторону главной камеры)
        baseRotation = transform.rotation; // а затем сохраняет полученную ориентацию в baseRotation
    }

    public override void Move() {
        float u = (Time.time - birthTime) / lifeTime; // Линейная интерполяция работает на основе значения u от 0 до 1
        if (u > 1) { // Если с момента birthTime прошло больше времени, чем задано в lifeTime
            Destroy(this.gameObject); // то значение u станет больше 1, и объект Enemy_2 будет уничтожен
            return; // Уничтожение Enemy_2 происходит по завершении интерполированного движения, а не в момент выхода объекта за пределы экрана
        }
        // Используйте AnimationCurve для установки поворота вокруг оси Y
        float shipRot = rotCurve.Evaluate(u) * 360; // Чтобы использовать AnimationCurve для преобразования значения, мы вызываем метод Evaluate(u) для исходного значения u (то есть значения u до применения кривой плавности типа «синусоида»). Полученное значение в диапазоне [0..1] затем умножается на 360, обеспечивая полный поворот на 360°
        // if (p0.x > p1.x) shipRot = -shipRot; // Если Enemy_2 движется справа налево (а не слева направо), измените направление вращения на противоположное, чтобы движение выглядело более естественным
        // transform.rotation = Quaternion.Euler(0, shipRot, 0);
        transform.rotation = baseRotation * Quaternion.Euler(-shipRot, 0, 0); // Здесь выполняется умножение baseRotation на новое вращение на угол -shipRot вокруг локальной оси X корабля
        // Отрегулируйте U-образную форму, добавив U-образную кривую на основе синусоидальной волны
        u = u + sinEccentricity * (Mathf.Sin(u * Mathf.PI * 2));
        // Интерполируйте две точки линейной интерполяции
        pos = (1 - u) * p0 + u * p1;
        
        // То есть смысл в чём, объект Enemy 2 появляется за границами экрана рандомно, выбирает рандомное движение по синусойде влево-вправо 
        // и исчезает через 10 сек примерно если его не уничтожает игрок. Плюс ко всему после добавления крена, перемножения кватерионов и изменения вращения корабля вокруг глобальной оси Х (baseRotation)
        // Enemy2 будет влетать в кадр, выполнять широкую петлю и улетать за пределы экрана
    }
}
```

## Скрипт Utils.cs

```c# title="C# Utils.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Utils : MonoBehaviour {

    //== Bézier Curves============================================================\\

    /// <summary>
    /// While most Bézier curves are 3 or 4 points, it is possible to have
    /// any number of points using this recursive function.
    /// </summary>
    /// <param name="u">The amount of interpolation [0..1]</param>
    /// <param name="points">An array of Vector3s to interpolate</param>
    static public Vector3 Bezier(float u, params Vector3[] points) {// Ключевое слово params позволяет передавать в метод либо массив, либо последовательность объектов Vector3
        // Создайте массив
        Vector3[,] vArr = new Vector3[points.Length, points.Length];
        // Заполните последнюю строку vArr элементами, представляющими собой точки
        int r = points.Length - 1;
        for (int c = 0; c < points.Length; c++) {
            vArr[r, c] = points[c];
        }

        // Пройдитесь по всем оставшимся строкам и интерполируйте точки в каждой из них 
        // Эти вложенные циклы for составляют основу данного быстрого метода построения кривой Безье. Циклы проходят по каждой пустой строке и выполняют интерполяцию точек на основе значений из вышележащей строки
        for (r--; r >= 0; r--) { // это уменьшает значение r с points.Length-1 до points.Length-2 еще до начала итераций. В случае с кривой Безье по трем точкам такой подход позволяет начать выполнение цикла со строки 1 массива vArr, используя значения из строки 2
            for (int c = 0; c <= r; c++) {
                vArr[r, c] = Vector3.LerpUnclamped(vArr[r + 1, c], vArr[r + 1, c + 1], u);
            }
        }
        // После завершения процесса vArr[0,0] содержит окончательное интерполированное значение
        return vArr[0, 0];
    }

    //== Materials Functions =======================================================\\
    /// <summary>
    /// Returns a list of all Materials on this GameObject and its children
    /// </summary>
    /// <param name="go">The GameObject on which to search for Renderers</param>
    static public Material[] GetAllMaterials(GameObject go) { // Будучи статическим публичным методом, GetAllMaterials() может быть вызван в любой части проекта с помощью конструкции Utils.GetAllMaterials()
        Renderer[] rends = go.GetComponentsInChildren<Renderer>(); // это метод, доступный для всех объектов GameObject; он перебирает сам объект и все его дочерние элементы, 
        // возвращая массив компонентов того типа, который был передан в качестве параметра шаблона <> (в данном примере - тип Renderer)
        Material[] mats = new Material[rends.Length];
        for (int i = 0; i < rends.Length; i++) { // Этот цикл for перебирает компоненты Renderer в массиве rends и извлекает поле material из каждого из них. Затем полученный материал добавляется в массив mats
            mats[i] = rends[i].material;
        }

        return mats;
    }
}
```

## Скрипт Enemy_3.cs

```c# title="C# Enemy_3.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Enemy_3 : Enemy {
    [Header("Enemy_3 Inscribed Fields")]
    public float lifeTime = 5;
    public Vector2 midpointYRange = new Vector2(1.5f, 3);
    [Tooltip("If true, the Bezier points & path are drawn in the Scene pane.")]
    public bool drawDebugInfo = true;

    [Header("Enemy_3 Private Fields")]
    [SerializeField]
    private Vector3[] points; // Три точки для кривой Безье
    [SerializeField]
    private float birthTime;

    void Start() { // Start хорошо работает, потому что он не используется в суперклассе Enemy
        points = new Vector3[3]; // Инициализация точек

        points[0] = pos;
        // Установите значения xMin и xMax так же, как это делает метод Main.SpawnEnemy()
        float xMin = -bndCheck.camWidth + bndCheck.radius;
        float xMax = bndCheck.camWidth - bndCheck.radius;
        // Выберите случайную центральную позицию в нижней половине экрана
        points[1] = Vector3.zero;
        points[1].x = Random.Range(xMin, xMax);
        float midYMult = Random.Range(midpointYRange[0], midpointYRange[1]); // Эти строки задают Y-координату первой (т. е. средней) точки кривой Безье в диапазоне от 1,5 до 3 расстояний от центра до нижней границы экрана, исходя из значения по умолчанию для Vector2 midpointYRange
        points[1].y = -bndCheck.camHeight * midYMult; // К компонентам .x и .y объекта Vector2 можно обращаться с помощью индексов (как [0] и [1] соответственно)
        // Выберите случайную конечную позицию в верхней части экрана
        points[2] = Vector3.zero;
        points[2].y = pos.y;
        points[2].x = Random.Range(xMin, xMax);
        // Установите время появления на текущее время
        birthTime = Time.time;

        if (drawDebugInfo) DrawDebug(); // Если переменная drawDebugInfo имеет значение true, здесь вызывается метод DrawDebug()
    }

    public override void Move() {
        // Кривые Безье работают на основе значения u от 0 до 1
        float u = (Time.time - birthTime) / lifeTime;
        if (u > 1) {
            Destroy(this.gameObject);
            return;
        }
        transform.rotation = Quaternion.Euler(u * 180, 0, 0); // Этот код применяет к кораблю глобальный поворот вокруг оси X, заставляя его перевернуться вблизи второй точки кривой Безье
        // Интерполируйте три точки кривой Безье 
        u = u - 0.1f * Mathf.Sin(u * Mathf.PI * 2); // Эффект плавности, чтобы улучшить движение по кривой Безье и уменьшить замедление в нижней части кривой
        pos = Utils.Bezier(u, points); // В этом вызове используется статический метод Utils.Bezier() из скрипта Utils
    }

    void DrawDebug() { // Этот метод отрисовывает линии, отображающие три вектора Vector3 из массива точек, 
        // а также саму кривую, по которой будет перемещаться игровой объект Enemy_3
        Debug.DrawLine(points[0], points[1], Color.cyan, lifeTime); // Рисует временную линию в окне Scene. Первые два параметра задают конечные точки линии, 
        Debug.DrawLine(points[1], points[2], Color.yellow, lifeTime); // третий - цвет, а последний - время (в секундах), в течение которого линия будет отображаться на экране, прежде чем исчезнет
        // Нарисуйте кривую Безье
        float numSections = 20;
        Vector3 prevPoint = points[0]; // Чтобы нарисовать саму кривую, мы разбиваем ее на 19 отдельных отрезков. Нулевая точка кривой сохраняется в переменной prevPoint
        Color col;
        Vector3 pt;
        for (int i = 1; i < numSections; i++) { // Цикл for начинается с индекса 1 (точка, соответствующая 1/20-й части или 5% длины кривой)
            float u = i / numSections; // В нем вычисляется вектор pt (точка на 5% пути вдоль кривой) 
            pt = Utils.Bezier(u, points); // с помощью метода Utils.Bezier()
            col = Color.Lerp(Color.cyan, Color.yellow, u); // а затем выбирается цвет c, соответствующий 5% перехода от голубого (cyan) к желтому (yellow)
            Debug.DrawLine(prevPoint, pt, col, lifeTime); // рисует линию от prevPoint до текущей точки pt, которая будет отображаться в течение времени lifeTime. 
            prevPoint = pt; // После этого значение pt присваивается переменной prevPoint, чтобы на следующей итерации цикла for линия начиналась в правильной точке
        }
    }
}
```

## Скрипт Weapon.cs

```c# title="C# Weapon.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// This is an enum of the various possible weapon types.
/// It also includes a "shield" type to allow a shield PowerUp.
/// Items marked [NI] below are Not Implemented in this book.
/// </summary>
public enum eWeaponType {
    none, // The default / no weapon
    blaster, // A simple blaster
    spread, // Multiple shots simultaneously
    phaser, // [NI] Shots that move in waves
    missile, // [NI] Homing missiles
    laser, // [NI] Damage over time
    shield // Raise shieldLevel
}

/// <summary>
/// The WeaponDefinition class allows you to set the properties
/// of a specific weapon in the Inspector. The Main class has
/// an array of WeaponDefinitions that makes this possible.
/// </summary>
[System.Serializable] // Атрибут [System.Serializable] делает класс, объявленный непосредственно после него, доступным для редактирования в инспекторе Unity
public class WeaponDefinition { // Вы можете изменять отдельные поля класса WeaponDefinition, чтобы корректировать характеристики пуль, выпускаемых вашим кораблем
    public eWeaponType type = eWeaponType.none;
    [Tooltip("Letter to show on the PowerUp Cube")]
    public string letter;
    [Tooltip("Color of PowerUp Cube")]
    public Color powerUpColor = Color.white; 
    [Tooltip("Prefab of Weapon model that is attached to the Player Ship")]
    public GameObject weaponModelPrefab;
    [Tooltip("Prefab of projectile that is fired")]
    public GameObject projectilePrefab;
    [Tooltip("Color of the Projectile that is fired")]
    public Color projectileColor = Color.white; 
    [Tooltip("Damage caused when a single Projectile hits an Enemy")]
    public float damageOnHit = 0;
    [Tooltip("Damage caused per second by the Laser [Not Implemented]")]
    public float damagePerSec = 0;
    [Tooltip("Seconds to delay between shots")]
    public float delayBetweenShots = 0;
    [Tooltip("Velocity of individual Projectiles")]
    public float velocity = 50;
}

public class Weapon : MonoBehaviour {
    static public Transform PROJECTILE_ANCHOR; // PROJECTILE_ANCHOR - это статический компонент Transform, созданный для выполнения роли 
    // родительского объекта в иерархии для всех игровых объектов ProjectileHero, создаваемых скриптами оружия

    [Header("Dynamic")] // Все три этих атрибута компилятора
    [SerializeField] // применяются к приватному полю eWeaponType _type
    [Tooltip("Setting this manually while playing does not work properly.")]
    private eWeaponType _type = eWeaponType.none;
    public WeaponDefinition def;
    public float nextShotTime;

    private GameObject weaponModel;
    private Transform shotPointTrans;

  [System.Obsolete] // Реализация метода Fire устарела
  void Start() {
        // Установите параметр PROJECTILE_ANCHOR, если он еще не установлен
        if (PROJECTILE_ANCHOR == null) { // Если PROJECTILE_ANCHOR равен null (поскольку он еще не был создан), данный скрипт создает новый игровой объект 
            GameObject go = new GameObject("_PojectileAnchor"); // с именем _ProjectileAnchor и присваивает его Transform переменной PROJECTILE_ANCHOR
            PROJECTILE_ANCHOR = go.transform;
        }

        shotPointTrans = transform.GetChild(0); // На момент вызова метода Start() у игрового объекта Weapon есть только один дочерний объект («ShotPoint»)
        // поэтому вместо вызова transform.Find("ShotPoint") мы можем просто получить нулевой дочерний объект из Transform игрового объекта Weapon

        SetType(_type); // Вызовите функцию SetType() для получения значения типа _type по умолчанию, заданного в инспекторе

        Hero hero = GetComponentInParent<Hero>(); // Найдите событие fireEvent компонента Hero в родительской иерархии
        if (hero != null) hero.fireEvent += Fire; // Eсли был найден экземпляр класса Hero, метод Fire() данного объекта Weapon добавляется к событию fireEvent в Hero
    }

    public eWeaponType type {
        get { return(_type); }
        set { SetType(value); }
    }

    public void SetType(eWeaponType wt) {
        _type = wt;
        if (type == eWeaponType.none) { // Если тип равен eWeaponType.none, данный GameObject переводится в неактивное состояние
            this.gameObject.SetActive(false); // Когда GameObject неактивен, он не получает вызовов методов MonoBehaviour (например, Update(), LateUpdate(), 
            return; // FixedUpdate(), OnCollisionEnter() и т. д.), не участвует в физической симуляции и визуально исчезает со сцены
        } else { // Однако вызов функций и изменение переменных в скриптах, прикрепленных к неактивному GameObject, по-прежнему возможны; поэтому, если что-либо вызовет SetType()
            this.gameObject.SetActive(true); // или установит свойство type в значение eWeaponType.blaster или eWeaponType.spread, метод SetType() сработает и повторно активирует GameObject, к которому он привязан
        }
        // Получите WeaponDefinition для этого типа оружия из Main
```

```c# title="C# Hero.cs"
        if (go == lastTriggerGo) return; // Если lastTriggerGo совпадает с go (текущим объектом GameObject, вызвавшим срабатывание триггера), то это событие игнорируется как дубликат, и выполнение функции прекращается

        lastTriggerGo = go; // Присвойте переменной lastTriggerGo значение go, чтобы подготовить ее к последующим вызовам OnTriggerEnter()
        Enemy enemy = go.GetComponent<Enemy>(); // Проверка наличия определенного скрипта на объекте
        PowerUp pUp = go.GetComponent<PowerUp>();
        if (enemy != null) { // Если щит был активирован enemy
            shieldLevel--; // Уменьшить уровень щита на 1
            Destroy(go); // … и уничтожить enemy
        } else if (pUp != null) { // Если щит задел усиление
            AbsorbPowerUp(pUp); // … поглотить PowerUp
        } else {
            Debug.LogWarning("Shield trigger hit by non-Enemy: " + go.name); // Если _Hero столкнется с объектом, не имеющим тега Enemy, в консоль будет выведено сообщение с информацией об этом объекте
        }
    }

    public void AbsorbPowerUp(PowerUp pUp) {
        Debug.Log("Absorbed PowerUp: " + pUp.type);
        switch(pUp.type) {
            case eWeaponType.shield: // Если бонус (PowerUp) относится к типу eWeaponType.shield, уровень щита повышается на 1
            shieldLevel++;
            break;
            default: // Любой другой тип eWeaponType для бонуса означает оружие; это случай, принимаемый по умолчанию
            if (pUp.type == weapons[0].type) { // Если тип eWeaponType бонуса совпадает с типом уже имеющегося оружия, выполняется поиск свободного слота для оружия и предпринимается попытка назначить этому слоту данный тип оружия. Если все пять слотов уже заняты, ничего не происходит
                Weapon weap = GetEmptyWeaponSlot();
                if (weap != null) {
                    weap.SetType(pUp.type); // Задать тип pUp
                }
            } else { // Если тип eWeaponType бонуса отличается, все слоты для оружия очищаются, а в слот weapons[0] устанавливается оружие нового типа, соответствующего подобранному бонусу
                ClearWeapons();
                weapons[0].SetType(pUp.type);
            }
            break;
        }
        pUp.AbsorbedBy(this.gameObject);
    }

    public float shieldLevel {
        get { return (_shieldLevel); }
        private set { // Ключевое слово private гарантирует, что, хотя получить значение shieldLevel может любой класс, изменить его способен только класс Hero
            _shieldLevel = Mathf.Min(value, 4); // Метод Mathf.Min() гарантирует, что значение _shieldLevel никогда не превысит 4
            
            if (value < 0) { // Если значение защиты будет установлено меньше нуля…
                Destroy(this.gameObject); // Destroy the Hero
                Main.HERO_DIED(); // И вызов метода HERO_DIED() из Main, который вызывает отложенный рестарт игры
            }
        }
    }

    /// <summary>
    /// Finds the first empty Weapon slot (i.e., type=none) and returns it.
    /// </summary>
    /// <returns>The first empty Weapon slot or null if none are empty</returns>
    Weapon GetEmptyWeaponSlot() {
        for (int i = 0; i < weapons.Length; i++) {
            if (weapons[i].type == eWeaponType.none) {
                return(weapons[i]);
            }
        }
        return(null);
    }
    /// <summary>
    /// Sets the type of all Weapon slots to none
    /// </summary>
    void ClearWeapons() {
        foreach (Weapon w in weapons) {
            w.SetType(eWeaponType.none);
        }
    }
}
```

```c# title="C# BlinkColorOnHit.cs"
using System.Collections.Generic;
using UnityEngine;

[DisallowMultipleComponent] // Атрибут компилятора [DisallowMultipleComponent] сообщает Unity, что каждый GameObject может содержать только один экземпляр этого скрипта
public class BlinkColorOnHit : MonoBehaviour {
    private static float blinkDuration = 0.1f; // Поскольку мы хотим, чтобы параметры blinkDuration и blinkColor были одинаковыми для всех экземпляров BlinkColorOnHit, логично сделать их закрытыми статическими полями

    private Material[] materials; 
    private Color[] originalColors;
    private BoundsCheck bndCheck;

        materials = Utils.GetAllMaterials( gameObject ); // Массив материалов заполняется с помощью нового метода Utils.GetAllMaterials()
        originalColors = new Color[materials.Length]; // Затем код перебирает все материалы и сохраняет их исходный цвет
        for (int i=0; i < materials.Length; i++) { // Хотя в данный момент все материалы игрового объекта Enemy имеют белый цвет, сохранение исходных цветов позволяет назначать им любой желаемый цвет
            originalColors[i] = materials[i].color; // менять его на blinkColor при получении врагом урона, а затем возвращать исходный цвет
        } // Важно отметить, что вызов Utils.GetAllMaterials() происходит в методе Awake(), а результат кэшируется в переменной materials. Это гарантирует, что операция выполняется лишь однажды для каждого врага
    }

        if (showingColor && Time.time > blinkCompleteTime) RevertColors(); // Если враг в данный момент окрашен (например, в красный цвет) и текущее время превышает blinkCompleteTime, вызывается метод RevertColors()

        // Проверьте наличие столкновений с ProjectileHero

        if (p != null) { // Как и в случае с получением урона в скрипте Enemy, если у игрового объекта, нанесшего удар, есть компонент ProjectileHero, мы попытаемся мигнуть красным цветом

            }
            SetColors(); // В противном случае вызывается метод SetColors()
        }
    }

    /// Sets the Albedo color (i.e., the main color) of all materials in the
    /// materials array to blinkColor, sets showingColor to true, and sets the
    /// time that the colors should be reverted.
    /// </summary>
    public void SetColors() {
        foreach (Material m in materials) {
            m.color = blinkColor;
        }
        showingColor = true;
        blinkCompleteTime = Time.time + blinkDuration;
    }

    /// <summary>
    /// Reverts all materials in the materials array back to their original color
    /// and sets showingColor to false.
    /// </summary>
    void RevertColors() {
        for (int i = 0; i < materials.Length; i++) {
            materials[i].color = originalColors[i];
        }
        showingColor = false;
    }
}
```

```c# title="C# PowerUp.cs"
        rigid.velocity = vel; // При умножении вектора на число с плавающей запятой (скаляр) изменяется длина вектора, но его направление остается прежним (или меняется на противоположное, если скаляр отрицателен)
        // В результате объект PowerUp получает скорость от 0,25 до 2 единиц в секунду (при использовании значений driftMinMax по умолчанию), направленную в случайную сторону в плоскости XY

            return;

            Color c = cubeMat.color;
            c.a = 1f - u; // Установите альфа-канал PowerCube на 1
            cubeMat.color = c;
            c = letter.color; // Букву тоже можно затушевать, но не так сильно
            c.a = 1f - (u * 0.5f); // Установите значение альфа-канала буквы равным 1-(u/2)
            letter.color = c;
        }
        if (!bndCheck.isOnScreen) { // Если бонус полностью исчез с экрана, уничтожьте его
            Destroy(gameObject);
        }
    }

    }
```

```c# title="C# Enemy_4.cs"
    // в момент вызова InitMovement()). Затем выбирается новая позиция p1; при этом используются данные компонента BoundsCheck, чтобы гарантировать, что точка находится в пределах экрана
        p0 = p1; // Установите p0 равным старому значению p1
        // Назначьте новое местоположение на экране для p1
        float widMinRad = bndCheck.camWidth - bndCheck.radius;
        float hgtMinRad = bndCheck.camHeight - bndCheck.radius;
        p1.x = Random.Range(-widMinRad, widMinRad);
        p1.y = Random.Range(-hgtMinRad, hgtMinRad);
        // Убедитесь, что оно перемещается в другой квадрант экрана
        if (p0.x * p1.x > 0 && p0.y * p1.y > 0) { // Поскольку центр экрана соответствует координатам [ 0, 0 ], положительные значения произведений 
        // p0.xp1.x и p0.yp1.y означают, что случайно выбранная на экране точка находится в том же квадранте, что и текущая позиция
            if (Mathf.Abs( p0.x) > Mathf.Abs(p0.y)) { // Чтобы перейти в другой квадрант, мы сравниваем удаленность текущей координаты x корабля от нуля с удаленностью координаты y:
                p1.x *= -1; // если x дальше от нуля, мы умножаем p1.x на -1, перемещая точку на противоположную сторону экрана; если дальше находится p0.y, мы инвертируем p1.y.
            } else { // Смена квадрантов значительно повышает вероятность того, что Enemy_4 будет совершать динамичные перемещения по экрану, а не ограничиваться незначительными сдвигами
                p1.y *= -1;
            }
        }
        // Сброс времени
        timeStart = Time.time;
    }

    public override void Move() {
        // Это полностью заменяет метод Enemy.Move() линейной интерполяцией
        float u = (Time.time -timeStart) / duration;
        if (u >= 1) {
            InitMovement();
            u = 0;
        }
        u = u - 0.15f * Mathf.Sin(u * 2 * Mathf.PI); // Эта строка применяет функцию плавности (easing) к значению u, заставляя корабль двигаться нелинейно
        // При использовании функции «Sine –0.15» корабль начинает движение медленно, ускоряется в середине пути и замедляется при приближении к p1. Благодаря этому создается ощущение массы корабля, а само движение выглядит более естественным.
        pos = (1 - u) * p0 + u * p1; // Эта строка выполняет простую линейную интерполяцию от p0 до p1
    }

    /// <summary>
    /// Enemy_4 Collisions are handled differently from other Enemy subclasses
    /// to enable protection by EnemyShields.
    /// </summary>
    /// <param name="coll"></param>
    void OnCollisionEnter(Collision coll) { 
        GameObject otherGO = coll.gameObject;
        
        ProjectileHero p = otherGO.GetComponent<ProjectileHero>(); // Проверка, что снаряд ProjectileHero попал
        if (p != null) {
            // Destroy the ProjectileHero regardless of bndCheck.isOnScreen
            Destroy(otherGO); 
            // Получение урона только при нахождении на экране
            if (bndCheck.isOnScreen) {
                // Поиск GameObject Enemy_4 по которому был нанесён урон
                GameObject hitGO = coll.contacts[0].thisCollider.gameObject; // Позволяет получить конкретный коллайдер, по которому было произведено попадание (thisCollider)
                if (hitGO == otherGO) { // Данная условная конструкция переназначает hitGO на otherCollider.gameObject
                    hitGO = coll.contacts[0].otherCollider.gameObject; // для того чтобы переменная hitGO ошибочно не указала на игровой объект ProjectileHero из-за ошибки движка (PhysX иногда путал thisCollider и otherCollider)
                }
                // Получите сумму урона из основного объекта WEAP_DICT
                float dmg = Main.GET_WEAPON_DEFINITION(p.type).damageOnHit;
                // Поиск EnemyShield по которому нанесён урон (если такое произошло)
                bool shieldFound = false;
                foreach (EnemyShield es in allShields) { // Этот цикл foreach перебирает игровые объекты всех компонентов EnemyShield и проверяет,  
                    if (es.gameObject == hitGO) { // является ли текущий объект тем самым, по которому был нанесен удар.
                        es.TakeDamage(dmg); // Если это так, у данного EnemyShield вызывается метод TakeDamage()
                        shieldFound = true;
                    }
                }
                if (!shieldFound) thisShield.TakeDamage(dmg); // В том маловероятном случае, если у пораженного игрового объекта отсутствует компонент EnemyShield, вызывается thisShield.TakeDamage(), что приводит к попытке нанести урон компоненту EnemyShield объекта Enemy_4
                // Если thisShield всё ещё активен, значит, он не был уничтожен
                if (thisShield.isActive) return; // После вызова TakeDamage() у соответствующего EnemyShield проверяется состояние здоровья всех компонентов EnemyShield: 
                // если оно исчерпано, переменная thisShield.isActive принимает значение false. Таким образом, если thisShield.isActive по-прежнему равно true, это означает, 
                // что у одного из компонентов EnemyShield в иерархии Enemy_4 еще осталось здоровье
                // Этот корабль уничтожен, нужно сделать запись в Main
                if (!calledShipDestroyed) {
                    Main.SHIP_DESTROYED(this);
                    calledShipDestroyed = true;
                }
                Destroy(gameObject);
            }
        } else {
            Debug.Log("Enemy_4 hit by non-ProjectileHero: " + otherGO.name);
        }
    }
}
```

```c# title="C# EnemyShield.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

[DisallowMultipleComponent]
[RequireComponent(typeof(BlinkColorOnHit))]
public class EnemyShield : MonoBehaviour {
    [Header("Inscribed")]
    public float health = 10;

    private List<EnemyShield> protectors = new List<EnemyShield>();
    private BlinkColorOnHit blinker;

    void Start() {
        blinker = GetComponent<BlinkColorOnHit>();
        blinker.ignoreOnCollisionEnter = true; 
        if ( transform.parent == null )
        return; // Эти строки сначала проверяют, есть ли у данного GameObject родительский объект в иерархии, а затем - есть ли у этого родителя 
        EnemyShield shieldParent = transform.parent.GetComponent<EnemyShield>(); // компонент EnemyShield. Если оба условия выполняются, то текущий EnemyShield регистрируется как защитник родительского объекта
        if (shieldParent != null) {
            shieldParent.AddProtector(this);
        }
    }

    /// <summary>
    /// Called by another EnemyShield to join the protectors of this EnemyShield
    /// </summary>
    /// <param name="shieldChild">The EnemyShield that will protect this</param>
    public void AddProtector(EnemyShield shieldChild) {
        protectors.Add(shieldChild);
    }
    /// <summary>
    /// Shortcut for gameObject.activeInHierarchy and gameObject.SetActive()
    /// </summary>
    public bool isActive {
        get {return gameObject.activeInHierarchy;} // Блок get свойства isActive является открытым (public), а блок set - закрытым (private); 
        private set {gameObject.SetActive(value);} // это означает, что получить значение свойства может любой класс, а изменить его - только класс EnemyShield
    }
    /// <summary>
    /// Called by Enemy_4.OnCollisionEnter() & parent’s EnemyShields.TakeDamage()
    /// to distribute damage to EnemyShield protectors.
    /// </summary>
    /// <param name="dmg">The amount of damage to be handled</param>
    /// <returns>Any damage not handled by this shield</returns>
    public float TakeDamage(float dmg) { // Если у данного EnemyShield есть защитники, он попытается перенаправить урон им;
        foreach (EnemyShield es in protectors) { // если же ни один из них не активен (или если необработанный урон возвращается обратно), то значение здоровья будет уменьшено
            if (es.isActive) {
                dmg = es.TakeDamage( dmg ); // Если какой-либо защитный компонент EnemyShield возвращает нулевой урон, то вверх по цепочке также 
                // передается ноль. Это позволяет избежать вызова метода TakeDamage() у других защитных компонентов
                if (dmg == 0) return 0; // Если весь ущерб был устранен, вернуть 0 единиц ущерба
            }
        }
        // Если код дойдёт до этого места, то этот вражеский щит мигнет и получит урон
        blinker.SetColors(); // Если выполнение кода не завершилось до этой строки, то данный EnemyShield мигнет, а затем получит урон
        health -= dmg;
        if (health <= 0) { // Значение dmg вычитается из health, и если результат меньше или равен нулю, данный EnemyShield (а также его GameObject) деактивируется
            isActive = false; // Деактивация EnemyShield
            return -health; // Возвращается значение health с обратным знаком, чтобы учесть необработанный урон. Например, если health равно 2, а dmg - 5, 
            // то операция health -= dmg приведет к тому, что health станет равным -3. Это означает, что 3 единицы урона не были поглощены данным EnemyShield, поэтому возвращается число 3 (то есть -health)
        }
        return 0; // Если выполнение кода доходит до этой строки, значит, весь урон был поглощен за счет health, и мы можем вернуть ноль
    }
}
```

```c# title="C# Parallax.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Parallax : MonoBehaviour {
    [Header("Inscribed")]
    public Transform playerTrans; // The player ship
    public Transform[] panels; // The scrolling foregrounds
    [Tooltip("Speed at which the panels move in Y")]
    public float scrollSpeed = -30f;
    [Tooltip("Controls how much panels react to player movement (Default 0.25)")]
    public float motionMult = 0.25f; // a

    private float panelHt; // Height of each panel
    private float depth; // Depth of panels (that is, pos.z)

    void Start () {
        panelHt = panels[0].localScale.y;
        depth = panels[0].position.z;

        // Set initial positions of panels
        panels[0].position = new Vector3(0,0,depth);
        panels[1].position = new Vector3(0,panelHt,depth);
    }

    void Update () {
        float tY, tX=0;
        tY= Time.time * scrollSpeed % panelHt + (panelHt *0.5f); // b

        if (playerTrans != null) {
            tX = -playerTrans.transform.position.x * motionMult; // c
            //tY += -poi.transform.position.y * motionMult;
        }

        // Position panels[0]
        panels[0].position = new Vector3(tX, tY, depth);
        // Position panels[1] where needed to make a continuous starfield // d
        if (tY >= 0) {
            panels[1].position = new Vector3(tX, tY -panelHt, depth);
        } else {
            panels[1].position = new Vector3(tX, tY +panelHt, depth);
        }
    }
}
```
