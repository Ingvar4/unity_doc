```c# title="C# Enemy_2.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Enemy_1 : Enemy { // Класс Enemy_1, являющийся расширением класса Enemy, наследует открытые (public) поля speed, fireRate, health и score, а также открытое свойство pos и открытый метод Move(). Однако он не наследует закрытое (private) поле bndCheck;
    [Header("Enemy_1 Inscribed Fields")]
    [Tooltip("# of seconds for a full sine wave")] // Этот атрибут компилятора [Tooltip()] указывает Unity отображать соответствующую подсказку, когда пользователь наводит курсор мыши на имя поля в окне Inspector
    public float waveFrequency = 2;
    [Tooltip("Sine wave width in meters")]
    public float waveWidth = 4;
    [Tooltip("Amount the ship will roll left and right with the sine wave")]
    public float waveRotY = 45;

    private float x0; // Начальное значение x в pos
    private float birthTime;

    void Start() {
        // Установить x0 равным начальной позиции x Enemy_1
        x0 = pos.x; // c

        birthTime = Time.time;
    }

    // Переопределяем функцию перемещения противника
    public override void Move() { // Метод является virtual в суперклассе, чтобы переопределить его тут, задаём тип override
        // Поскольку pos является свойством, вы не можете напрямую задать pos.x
        // так что получите значение pos в виде редактируемого Vector3
        Vector3 tempPos = pos;
        // Тета-ритм изменяется в зависимости от времени
        float age = Time.time - birthTime;
        float theta = Mathf.PI * 2 * age / waveFrequency;
        float sin = Mathf.Sin(theta);
        tempPos.x = x0 + waveWidth * sin;
        pos = tempPos; // немного повернуть вокруг оси Y
        Vector3 rot = new Vector3(0, sin *waveRotY, 0);
        this.transform.rotation = Quaternion.Euler(rot);

        // Метод base.Move() по-прежнему обрабатывает перемещение вниз по оси Y
        base.Move(); // Вызов base.Move() обращается к методу Move() суперкласса Enemy. 
        // В данном случае метод Move() в подклассе Enemy_1 отвечает за горизонтальное перемещение по синусоиде, 
        // тогда как метод Move() в суперклассе Enemy перемещает игровой объект (GameObject) Enemy_1 вниз по экрану

        // print( bndCheck.isOnScreen ); // Оставьте эту строку закомментированной
    }
}
```

```c# title="C# Enemy_2.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Enemy_2 : Enemy { // Класс Enemy_2 также наследует класс Enemy
    [Header("Enemy_2 Inscribed Fields")]
    public float lifeTime = 10;
    // Enemy_2 использует синусоидальный сигнал для модификации двухточечной линейной интерполяции
    [Tooltip("Определяет, насколько синусоида облегчит интерполяцию")]
    public float sinEccentricity = 0.6f;
    public AnimationCurve rotCurve; // Это добавляет публичное поле типа AnimationCurve, которое можно редактировать в окне Inspector

    [Header("Enemy_2 Private Fields")]
    [SerializeField] private float birthTime; // Время начала интерполяции // SerializeField заставляет Unity отображать данное поле в окне Inspector (и позволяет редактировать его там же), несмотря на то, что поле является закрытым (private)
    private Quaternion baseRotation; // Кватернион baseRotation хранит ориентацию объекта Enemy_2, когда он находится в точке p0 и смотрит (то есть направлен) на точку p1
    [SerializeField] private Vector3 p0, p1; // Lerp_points

    void Start() {
        // Выберите любую точку в левой части экрана
        // В этом фрагменте кода выбирается случайная точка у левого края экрана. Сначала выбирается координата X, 
        // находящаяся непосредственно за пределами левой границы экрана: значение -bndCheck.camWidth соответствует 
        // левому краю, а вычитание bndCheck.radius гарантирует, что Enemy_2 будет полностью скрыт за пределами экрана 
        // (координата X смещается влево на величину, равную радиусу объекта Enemy_2). Затем выбирается случайная 
        // координата Y в диапазоне между нижней границей экрана (- bndCheck.camHeight) и верхней границей (bndCheck.camHeight)
        p0 = Vector3.zero;
        p0.x = -bndCheck.camWidth - bndCheck.radius;
        p0.y = Random.Range( -bndCheck.camHeight, bndCheck.camHeight );

        // Выберите любую точку в правой части экрана
        p1 = Vector3.zero; // Аналогичным образом выбирается конечная точка за пределами правой границы экрана.
        p1.x = bndCheck.camWidth + bndCheck.radius;
        p1.y = Random.Range( -bndCheck.camHeight, bndCheck.camHeight );

        // Возможная смена сторон
        if (Random.value > 0.5f) {
            // Установка отрицательного значения .x для каждой точки переместит её на 
            // другую сторону экрана
            p0.x *= -1;
            p1.x *= -1;
        }
        // Установите значение поля birthTime равным текущему времени
        birthTime = Time.time; // Переменная birthTime используется для интерполяции в функции Move().
        // Настройте начальный поворот корабля
        transform.position = p0;  // Этот фрагмент кода размещает Enemy_2 в точке p0, разворачивает его в сторону p1, а затем сохраняет полученную ориентацию в baseRotation
        transform.LookAt(p1, Vector3.back); // Вызов transform.LookAt(p1, Vector3.back) направляет ось transform.forward данного объекта на точку p1. 
        // Второй параметр, Vector3.back, указывает методу LookAt(), что после поворота на p1 локальная ось Y (ось «вверх») объекта должна быть направлена 
        // ​​назад вдоль глобальной оси -Z (то есть в сторону главной камеры)
        baseRotation = transform.rotation;
    }

    public override void Move() {
        // Линейная интерполяция работает на основе значения u от 0 до 1
        float u = (Time.time - birthTime) / lifeTime;
        // Если u>1, то прошло больше времени, чем lifeTime с момента рождения.
        if (u > 1) {
            // Этот Враг_2 завершил свою жизнь
            Destroy( this.gameObject ); // Если с момента birthTime прошло больше времени, чем задано в lifeTime, то значение u станет больше 1, и объект Enemy_2 будет уничтожен. Уничтожение Enemy_2 происходит по завершении интерполированного движения, а не в момент выхода объекта за пределы экрана
            return;
        }

        // Используйте AnimationCurve для установки поворота вокруг оси Y
        float shipRot = rotCurve.Evaluate( u ) * 360; // Чтобы использовать AnimationCurve для преобразования значения, 
        // мы вызываем метод Evaluate(u) для исходного значения u (то есть значения u до применения кривой плавности типа «синусоида»). 
        // Полученное значение в диапазоне [0..1] затем умножается на 360, обеспечивая полный поворот на 360°
        // if ( p0.x > p1.x ) shipRot = -shipRot; // Если Enemy_2 движется справа налево (а не слева направо), измените направление вращения на противоположное, чтобы движение выглядело более естественным
        // transform.rotation = Quaternion.Euler(0,shipRot,0);
        transform.rotation = baseRotation * Quaternion.Euler(-shipRot, 0, 0); // Здесь выполняется умножение baseRotation (сохраненного на этапе // b) на новое вращение на угол -shipRot вокруг локальной оси X корабля. Мы обсудим это в ближайшее время

        // Отрегулируйте u, добавив U-образную кривую на основе синусоидальной волны.
        u = u + sinEccentricity * (Mathf.Sin(u * Mathf.PI * 2));
        // Интерполируем две точки линейной интерполяции
        pos = (1-u)*p0 + u*p1;
        // Обратите внимание, что Enemy_2 НЕ вызывает метод base.Move()
    }
}
```

## Скрипт Utils.cs

```c# title="C# Utils.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Utils : MonoBehaviour { // Хотя у текущей версии класса Utils нет причин наследоваться от MonoBehaviour, для последующих дополнений потребуется этот базовый класс

    //== Bézier Curves============================================================\\

    /// <summary>
    /// Хотя большинство кривых Безье имеют 3 или 4 точки, с помощью этой рекурсивной функции можно получить
    /// любое количество точек.
    /// </summary>
    /// <param name="u">Величина интерполяции [0..1]</param>
    /// <param name="points">Массив векторов Vector3 для интерполяции</param>
    static public Vector3 Bezier( float u, params Vector3[] points ) {// Ключевое слово params позволяет передавать в метод либо массив, либо последовательность объектов Vector3
        // Создайте массив
        Vector3[,] vArr = new Vector3[points.Length, points.Length];
        // Заполните последнюю строку vArr элементами из points
        int r = points.Length - 1;
        for ( int c = 0; c < points.Length; c++ ) {
            vArr[r, c] = points[c];
        }

        // Пройдитесь по всем оставшимся строкам и интерполируйте точки в каждой из них // c
        for ( r--; r >= 0; r-- ) { // Эти вложенные циклы for составляют основу данного быстрого метода построения кривой Безье. 
        // Циклы проходят по каждой пустой строке и выполняют интерполяцию точек на основе значений из вышележащей строки
            for ( int c = 0; c <= r; c++ ) {
                vArr[r,c] = Vector3.LerpUnclamped(vArr[r + 1, c],
                vArr[r + 1, c + 1], u);
            }
        }
        // Обратите внимание на использование выражения r-- в блоке инициализации цикла for: это уменьшает 
        // значение r с points.Length-1 до points.Length-2 еще до начала итераций. В случае с кривой Безье по трем 
        // точкам такой подход позволяет начать выполнение цикла со строки 1 массива vArr, используя значения из строки 2

        // После завершения процесса vArr[0,0] содержит окончательное интерполированное значение
        return vArr[0, 0];
    }

    //== Materials Functions =======================================================\\
    /// <summary>
    /// Returns a list of all Materials on this GameObject and its children
    /// </summary>
    /// <param name="go">The GameObject on which to search for Renderers</param>
    static public Material[] GetAllMaterials( GameObject go ) { // Будучи статическим публичным методом, GetAllMaterials() может быть вызван в любой части проекта с помощью конструкции Utils.GetAllMaterials()
        Renderer[] rends = go.GetComponentsInChildren<Renderer>(); // GetComponentsInChildren<>() - это метод, доступный для всех объектов GameObject; он перебирает сам объект и все его дочерние элементы, возвращая массив компонентов того типа, который был передан в качестве параметра шаблона <> (в данном примере - тип Renderer)
        Material[] mats = new Material[rends.Length];
        for (int i = 0; i < rends.Length; i++) { // Перебирает компоненты Renderer в массиве rends и извлекает поле material из каждого из них. Затем полученный материал добавляется в массив mats
            mats[i] = rends[i].material;
        }
        return mats;
    }
}
```

```c# title="C# Enemy_3.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Enemy_3 : Enemy { // Класс Enemy_3 также является производным от класса Enemy
    [Header( "Enemy_3 Inscribed Fields" )]
    public float lifeTime = 5;
    public Vector2 midpointYRange = new Vector2( 1.5f, 3 );

    public bool drawDebugInfo = true;

    [Header( "Enemy_3 Private Fields" )]
    [SerializeField]
    private Vector3[] points; // Три точки для кривой Безье

    private float birthTime;

  // Опять же, метод Start хорошо работает, потому что он не используется в суперклассе Enemy

        points = new Vector3[3]; // Инициализация точек

        // Начальная позиция уже установлена ​​функцией Main.SpawnEnemy() 
        points[0] = pos;

        // Устанавливаем значения xMin и xMax так же, как это делает Main.SpawnEnemy()
        float xMin = -bndCheck.camWidth + bndCheck.radius;
        float xMax = bndCheck.camWidth - bndCheck.radius;

        // Выберите случайную центральную позицию в нижней половине экрана
        points[1] = Vector3.zero;
        points[1].x = Random.Range( xMin, xMax );
        float midYMult = Random.Range(midpointYRange[0], midpointYRange[1]); 
        points[1].y = -bndCheck.camHeight * midYMult; 
        // Эти строки задают Y-координату первой (т. е. средней) точки кривой Безье в диапазоне от 1,5 до 3 расстояний 
        // от центра до нижней границы экрана, исходя из значения по умолчанию для Vector2 midpointYRange. Обратите внимание: 
        // к компонентам .x и .y объекта Vector2 можно обращаться с помощью индексов (как [0] и [1] соответственно)

        // Выберите случайную конечную позицию в верхней части экрана
        points[2] = Vector3.zero;
        points[2].y = pos.y;
        points[2].x = Random.Range( xMin, xMax );

        // Установите время появления на текущее время
        birthTime = Time.time;

        if ( drawDebugInfo ) DrawDebug(); // Если переменная drawDebugInfo имеет значение true, здесь вызывается метод DrawDebug()
    }

    public override void Move() {
        // Кривые Безье работают на основе значения u от 0 до 1
        float u = ( Time.time - birthTime ) / lifeTime;
        if ( u > 1 ) {
            // Enemy_3 завершил свою жизнь
            Destroy( this.gameObject );
            return;
        }
        transform.rotation = Quaternion.Euler( u * 180, 0, 0 ); // Этот код применяет к кораблю глобальный поворот вокруг оси X, заставляя его перевернуться вблизи второй точки кривой Безье
        // Интерполируйте три точки кривой Безье
        u = u - 0.1f * Mathf.Sin( u * Mathf.PI * 2 );
        pos = Utils.Bezier( u, points ); // В этом вызове используется статический метод Utils.Bezier(), который вы только что написали
        // Enemy_3 does not call base.Move()
    }
    void DrawDebug() { // Этот метод отрисовывает линии, отображающие три вектора Vector3 из массива точек, 
    // а также саму кривую, по которой будет перемещаться игровой объект Enemy_3. Вам стоит увидеть линии, 
    // отрисованные этой функцией, еще до чтения пояснений к ней, поэтому я вернусь к пунктам // f и // g через несколько шагов
        // Нарисуйте три точки
        Debug.DrawLine( points[0], points[1], Color.cyan, lifeTime ); // Метод Debug.DrawLine() рисует временную линию в окне Scene. 
        // Первые два параметра задают конечные точки линии, третий - цвет, а последний - время (в секундах), в течение которого линия 
        // будет отображаться на экране, прежде чем исчезнет. Установив длительность равной lifeTime, мы можем нарисовать линию один раз 
        // и быть уверенными, что она исчезнет ровно в тот момент, когда Enemy_3 завершит свое движение
        Debug.DrawLine( points[1], points[2], Color.yellow, lifeTime );
        // Нарисуйте кривую Безье
        float numSections = 20;
        Vector3 prevPoint = points[0]; // Чтобы нарисовать саму кривую, мы разбиваем ее на 19 отдельных отрезков. Нулевая точка кривой сохраняется в переменной prevPoint
        Color col;
        Vector3 pt;
        for ( int i = 1; i < numSections; i++ ) { // Цикл for начинается с индекса 1 (точка, соответствующая 1/20-й части или 5% длины кривой). 
        // В нем вычисляется вектор pt (точка на 5% пути вдоль кривой) с помощью метода Utils.Bezier(), а затем выбирается цвет c, 
        // соответствующий 5% перехода от голубого (cyan) к желтому (yellow)
            float u = i / numSections;
            pt = Utils.Bezier(u, points);
            col = Color.Lerp( Color.cyan, Color.yellow, u );
            Debug.DrawLine( prevPoint, pt, col, lifeTime ); // Метод Debug.DrawLine() рисует линию от prevPoint до текущей точки pt, 
            // которая будет отображаться в течение времени lifeTime. После этого значение pt присваивается переменной prevPoint, 
            // чтобы на следующей итерации цикла for линия начиналась в правильной точке
            prevPoint = pt;
        }
    }
}
```

```c# title="C# Weapon.cs"
        // данный скрипт создает новый игровой объект с именем _ProjectileAnchor и присваивает его Transform переменной PROJECTILE_ANCHOR
        if (PROJECTILE_ANCHOR == null) {
            GameObject go = new GameObject("_ProjectileAnchor");
            PROJECTILE_ANCHOR = go.transform;
        }
        // Мы знаем, что на момент вызова метода Start() у игрового объекта оружия есть только один дочерний объект («ShotPoint»), 
        // поэтому вместо вызова transform.Find("ShotPoint") мы можем просто получить нулевой дочерний объект из Transform игрового объекта оружия. 
        // (Этот подход меняет проблему возможной опечатки в имени «ShotPoint» (в самом объекте или в коде) на проблему того, что произойдет, если в префаб игрового объекта оружия будет добавлен еще один дочерний объект.)
        shotPointTrans = transform.GetChild( 0 );

        // Вызовите функцию SetType() для получения значения типа _type по умолчанию, заданного в инспекторе
        // При запуске класса Weapon вызывается метод SetType() с текущим значением поля eWeaponType _type. Это гарантирует, 
        // что игровой объект оружия либо исчезнет (если _type равен eWeaponType.none), либо отобразит правильную модель оружия 
        // (если _type равен eWeaponType.blaster или eWeaponType.spread). Ручное изменение поля _type во время работы игры не дает нужного результата, 
        // так как при этом пропускается вызов SetType(); однако метод SetType() вызывается через блок set свойства type, поэтому присвоение нового значения eWeaponType свойству type обеспечит назначение правильной модели игровому объекту
        SetType( _type ); 

        // Найдите событие fireEvent компонента Hero в родительской иерархии
        Hero hero = GetComponentInParent<Hero>(); // В рамках метода Start() эта строка выполняет поиск вверх по иерархии (над данным игровым объектом оружия) 
        // компонента класса Hero, прикрепленного к одному из родительских игровых объектов в Unity. Этот поиск должен всегда возвращать экземпляр класса Hero, прикрепленный к игровому объекту _Hero на сцене
        if ( hero != null ) hero.fireEvent += Fire;
        // Если был найден экземпляр класса Hero, метод Fire() данного объекта Weapon добавляется к событию fireEvent этого экземпляра. 
        // В дальнейшем событие fireEvent класса Hero сможет вызывать метод Fire() данного объекта Weapon. Вас может удивить, что это возможно, 
        // ведь метод Fire() является закрытым (private). Однако скрипт Weapon имеет доступ к Fire() и поэтому может добавить этот метод к событию fireEvent
    }

    public eWeaponType type {
        get { return( _type ); }
        set { SetType( value ); }
    }

    public void SetType( eWeaponType wt ) {
        _type = wt;
        // Если тип равен eWeaponType.none, данный GameObject переводится в неактивное состояние. Когда GameObject неактивен, 
        // он не получает вызовов методов MonoBehaviour (например, Update(), LateUpdate(), FixedUpdate(), OnCollisionEnter() и т. д.), 
        // не участвует в физической симуляции и визуально исчезает со сцены. Однако вызов функций и изменение переменных в скриптах, 
        // прикрепленных к неактивному GameObject, по-прежнему возможны; поэтому, если что-либо вызовет SetType() или установит свойство 
        // type в значение eWeaponType.blaster или eWeaponType.spread, метод SetType() сработает и повторно активирует GameObject, к которому он привязан
        if (type == eWeaponType.none) { // f
            this.gameObject.SetActive(false);
            return;
        } else {
            this.gameObject.SetActive(true);
        }
        // Получите WeaponDefinition для этого типа оружия из Main
        def = Main.GET_WEAPON_DEFINITION(_type);
        // Уничтожьте любую старую модель, а затем прикрепите модель для этого оружия
        // Эти строки извлекают соответствующий объект WeaponDefinition из Main и создают экземпляр weaponModel как дочерний 
        // объект для данного GameObject (оружия). Если weaponModel уже был прикреплен, он предварительно уничтожается. 
        // Параметры localPosition и localScale настраиваются так, чтобы обеспечить правильное выравнивание модели
        if ( weaponModel != null ) Destroy( weaponModel );
        weaponModel = Instantiate<GameObject>(def.weaponModelPrefab, transform);
        weaponModel.transform.localPosition = Vector3.zero;
        weaponModel.transform.localScale = Vector3.one;

        nextShotTime = 0; // Вы можете запустить процесс стрельбы сразу после установки параметра _type 
        // Установка nextShotTime в значение 0 позволяет произвести выстрел из этого оружия немедленно
    }

  [System.Obsolete]
  private void Fire() {
        // Если this.gameObject неактивен, то возврат
        if ( !gameObject.activeInHierarchy ) return; // Свойство gameObject.activeInHierarchy будет иметь значение false, если сам GameObject оружия неактивен либо если GameObject _Hero (корневой родительский объект для этого оружия) неактивен или уничтожен
        // Если между выстрелами прошло недостаточно времени, то возврат
        if ( Time.time < nextShotTime ) return; // Если текущее время меньше nextShotTime, выполнение метода прекращается, и выстрел не производится. Это предотвращает слишком частую стрельбу

        ProjectileHero p;
        Vector3 vel = Vector3.up * def.velocity;

        // Оператор switch содержит варианты для каждого из двух типов оружия (WeaponType). 
        // Тип eWeaponType.blaster создает один GameObject типа ProjectileHero путем вызова MakeProjectile() (который возвращает ссылку 
        // на экземпляр класса ProjectileHero, прикрепленный к новому GameObject снаряда), а затем задает скорость (velocity) его компоненту Rigidbody в направлении вектора vel.
        switch (type) { 
            case eWeaponType.blaster:
            p = MakeProjectile();
            p.vel = vel;
            break;

            // Если _type равен eWeaponType.spread, создаются три различных снаряда (Projectile). Направление двух из них поворачивается 
            // на 10 градусов вокруг оси Vector3.back (то есть оси -Z, направленной от экрана в вашу сторону). Затем свойству 
            // Rigidbody.velocity присваивается результат умножения этого кватерниона (вращения) на вектор vel. При умножении кватерниона на 
            // Vector3 происходит поворот этого вектора, в результате чего итоговый вектор скорости оказывается направленным туда, куда ориентирован объект ProjectileHero
            case eWeaponType.spread: 
            p = MakeProjectile();
            p.vel = vel;
            p = MakeProjectile();
            p.transform.rotation = Quaternion.AngleAxis( 10, Vector3.back );
            p.vel = p.transform.rotation * vel;
            p = MakeProjectile();
            p.transform.rotation = Quaternion.AngleAxis(-10, Vector3.back );
            p.vel = p.transform.rotation * vel;
            break;
        }
    }
    private ProjectileHero MakeProjectile() { // Метод MakeProjectile() создает экземпляр (клонирует) префаба, хранящегося в WeaponDefinition, и возвращает ссылку на прикрепленный к нему экземпляр класса ProjectileHero
        GameObject go;
        // Игровой объект ProjectileHero создается как дочерний объект PROJECTILE_ANCHOR с использованием второго параметра 
        // метода Instantiate(); это эффективнее, чем назначать родительский объект для transform позднее. Благодаря этому объект 
        // размещается в иерархии сцены (Hierarchy) внутри _ProjectileAnchor, что позволяет избежать загромождения списка иерархии множеством клонов ProjectileHero
        go = Instantiate<GameObject>(def.projectilePrefab,PROJECTILE_ANCHOR);
        ProjectileHero p = go.GetComponent<ProjectileHero>();

        Vector3 pos = shotPointTrans.position;
        pos.z = 0; // Координата Z объекта ProjectileHero устанавливается равной 0, чтобы учесть поворот корабля Hero и любые другие факторы, способные сместить объект относительно плоскости Z=0
        p.transform.position = pos;

        p.type = type;
        nextShotTime = Time.time + def.delayBetweenShots; // Переменной nextShotTime присваивается значение, равное текущему времени плюс def.delayBetweenShots; 
        // это предотвращает стрельбу из данного оружия в течение def.delayBetweenShots секунд
        return( p );
    }
}
```

```c# title="C# Main.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.SceneManagement; // Обеспечивает загрузку и перезагрузку сцен

public class Main : MonoBehaviour {
    static private Main S; // Частный синглтон для Main
    static private Dictionary<eWeaponType, WeaponDefinition> WEAP_DICT; // При объявлении и определении словарей указываются типы ключа и значения. 
    // Объявление WEAP_DICT как static и private означает, что доступ к нему могут получить любой экземпляр класса Main и любой статический метод этого класса, однако для других классов он недоступен.

    [Header("Inscribed")]
    public bool spawnEnemies = true; // В верхней части инспектора для основного скрипта (Main (Script)) появится флажок spawnEnemies
    public GameObject[] prefabEnemies; // Массив префабов Enemy 
    public float enemySpawnPerSecond = 0.5f; // Спавн врагов в сек.
    public float enemyInsetDefault = 1.5f; // Inset from the sides
    public float gameRestartDelay = 2; 
    public GameObject prefabPowerUp; // Здесь будет храниться префаб для всех бонусов (PowerUps)

    public eWeaponType[] powerUpFrequency = new eWeaponType[] { // Определяет частоту появления каждого типа бонуса (PowerUp)
        eWeaponType.blaster,
        eWeaponType.blaster,
        eWeaponType.spread, eWeaponType.shield
    };

    private BoundsCheck bndCheck;

    void Awake() {
        S = this;
        bndCheck = GetComponent<BoundsCheck>(); // Установите компонент bndCheck так, чтобы он ссылался на компонент BoundsCheck для этого GameObject
        Invoke( nameof(SpawnEnemy), 1f / enemySpawnPerSecond ); // Вызывает функцию SpawnEnemy() один раз (через 2 секунды, исходя из значений по умолчанию)

        // Универсальный словарь, в котором eWeaponType является ключом
        WEAP_DICT = new Dictionary<eWeaponType, WeaponDefinition>(); // Инициализация словаря
        foreach ( WeaponDefinition def in weaponDefinitions ) { // Этот цикл перебирает все элементы массива weaponDefinitions и создает соответствующую запись в словаре WEAP_DICT
            WEAP_DICT[def.type] = def;
        }
    }

    public void SpawnEnemy() {
        // Если значение spawnEnemies равно false, переходим к следующему вызову функции SpawnEnemy()
        if ( !spawnEnemies ) { // Использование метода Invoke() в данном случае позволяет нам впоследствии возобновить появление врагов, установив для spawnEnemies значение true
            Invoke( nameof( SpawnEnemy ), 1f / enemySpawnPerSecond );
            return;
        }

        // Выберите случайный префаб врага для создания экземпляра
        int ndx = Random.Range(0, prefabEnemies.Length); // В зависимости от длины массива prefabEnemies, выбирается случайное число, от 0 до единицы меньше, чем prefabEnemies.Length. Таким образом, если массив prefabEnemies содержит четыре врага, вызов Random.Range() вернет 0, 1, 2 или 3
        GameObject go = Instantiate<GameObject>( prefabEnemies[ ndx ] ); // Сгенерированный случайный ndx используется для выбора префаба GameObject из prefabEnemies
        // Разместите врага над экраном в произвольном месте по оси X
        float enemyInset = enemyInsetDefault; // EnemyInset изначально устанавливается равным enemyInsetDefault, заданному в Инспекторе
        if (go.GetComponent<BoundsCheck>() != null) { 
            enemyInset = Mathf.Abs( go.GetComponent<BoundsCheck>().radius );
            // Однако, если выбранный префаб врага имеет компонент BoundsCheck, вы вместо этого считываете радиус из него. 
            // Берется абсолютное значение радиуса, потому что иногда радиус устанавливается на отрицательное значение, 
            // так что GameObject должен полностью находиться за пределами экрана, прежде чем он будет считаться находящимся за пределами экрана, как в случае с Enemy_0
        }
        // Задайте начальное положение для появившегося врага
        // Используется BoundsCheck этого GameObject _MainCamera для получения camWidth и camHeight и выбирает позицию X, 
        // где созданный враг полностью находится на экране горизонтально. EnemyInset используется для того, чтобы убедиться, 
        // что враг не появляется частично за пределами экрана
        // Затем выбирается позиция по оси Y, где враг находится чуть выше экрана.
        Vector3 pos = Vector3.zero;
        float xMin = -bndCheck.camWidth + enemyInset;
        float xMax = bndCheck.camWidth - enemyInset;
        pos.x = Random.Range( xMin, xMax );
        pos.y = bndCheck.camHeight + enemyInset;
        go.transform.position = pos;
        // Снова вызвать SpawnEnemy()
        Invoke( nameof(SpawnEnemy), 1f/enemySpawnPerSecond ); 
    }

    void DelayedRestart() { // Метод DelayedRestart() вызывает Restart() через две секунды
        Invoke( nameof(Restart), gameRestartDelay );
    }
    void Restart() {
        // Перезагрузите __Scene_0, чтобы перезапустить игру
        SceneManager.LoadScene( "__Scene_0" ); // d
    }
    static public void HERO_DIED() {
        S.DelayedRestart(); // Любой другой класс может вызвать этот статический публичный метод HERO_DIED() 
        // через Main.HERO_DIED(). В свою очередь, HERO_DIED() вызовет DelayedRestart() у приватного синглтона S класса Main
    }

    /// <summary>
    /// Static function that gets a WeaponDefinition from the WEAP_DICT static
    /// protected field of the Main class.
    /// </summary>
    /// <returns>The WeaponDefinition, or if there is no WeaponDefinition with
    /// the eWeaponType passed in, returns a new WeaponDefinition with a
    /// eWeaponType of eWeaponType.none.</returns>
    /// <param name="wt">The eWeaponType of the desired WeaponDefinition</param>
    static public WeaponDefinition GET_WEAPON_DEFINITION( eWeaponType wt ) { 
        if (WEAP_DICT.ContainsKey(wt)) { // Это условное выражение проверяет, есть ли у WEAP_DICT запись с ключом, который был передан как wt
            return( WEAP_DICT[wt]);
        }
        // Если в WEAP_DICT нет записи нужного типа, вернуть
        // новый объект WeaponDefinition с типом eWeaponType.none (значение по умолчанию)
        return( new WeaponDefinition() );
    }

    /// <summary>
    /// Called by an Enemy ship whenever it is destroyed. It sometimes creates
    /// a PowerUp in place of the destroyed ship.
    /// </summary>
    /// <param name="e">The Enemy that was destroyed</param>
    static public void SHIP_DESTROYED( Enemy e ) {
        // Potentially generate a PowerUp
        if (Random.value <= e.powerUpDropChance) {  
            // Choose a PowerUp from the possibilities in powerUpFrequency
            int ndx = Random.Range( 0, S.powerUpFrequency.Length ); // Выбор случайного элемента массива

            // Spawn a PowerUp
            GameObject go = Instantiate<GameObject>( S.prefabPowerUp );
            PowerUp pUp = go.GetComponent<PowerUp>();
            // Set it to the proper WeaponType
            pUp.SetType( pUpType ); 
            // Set it to the position of the destroyed ship
            pUp.transform.position = e.transform.position;
        }
    }
}
```

```c# title="C# PowerUp.cs"
    public TextMesh letter; // Reference to the TextMesh
    public Vector3 rotPerSecond; // Euler rotation speed for PowerCube

        letter = GetComponent<TextMesh>();
        rigid = GetComponent<Rigidbody>();
        bndCheck = GetComponent<BoundsCheck>();
        cubeMat = cube.GetComponent<Renderer>().material;

        // Set a random velocity
        Vector3 vel = Random.onUnitSphere; // Get Random XYZ velocity // c
        vel.z = 0; // Flatten the vel to the XY plane
        vel.Normalize(); // Normalizing a Vector3 sets its length to 1m

        vel *= Random.Range(driftMinMax.x, driftMinMax.y); // d
        rigid.velocity = vel;

        // Set the rotation of this PowerUp GameObject to R:[ 0, 0, 0 ]
        transform.rotation = Quaternion.identity;
        // Quaternion.identity is equal to no rotation.

        // Randomize rotPerSecond for PowerCube using rotMinMax x & y // e
        rotPerSecond = new Vector3( Random.Range( rotMinMax[0], rotMinMax[1] ),
        Random.Range( rotMinMax[0], rotMinMax[1] ),
        Random.Range( rotMinMax[0], rotMinMax[1] ) );

        birthTime = Time.time;
    }

        cube.transform.rotation = Quaternion.Euler( rotPerSecond*Time.time ); // f

        // Given the default values, a PowerUp will exist for 10 seconds
        // and then fade out over 4 seconds.
        float u = (Time.time - (birthTime + lifeTime)) / fadeTime; // g
        // If u >= 1, destroy this PowerUp
        if (u >= 1) {
            Destroy( this.gameObject );
            return;
        }
        // If u>0, decrease the opacity (i.e., alpha) of the PowerCube & Letter
        if (u > 0) {
            Color c = cubeMat.color;
            c.a = 1f - u; // Set the alpha of PowerCube to 1-u
            cubeMat.color = c;
            // Fade the Letter too, just not as much
            c = letter.color;
            c.a = 1f - (u * 0.5f); // Set the alpha of the letter to 1-(u/2)
            letter.color = c;
        }

        if (!bndCheck.isOnScreen) {
            // If the PowerUp has drifted entirely off screen, destroy it
            Destroy( gameObject );
        }
```

```c# title="C# Enemy_4.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

[RequireComponent( typeof(EnemyShield) )]
public class Enemy_4 : Enemy { // Enemy_4 also extends the Enemy class
    [Header("Enemy_4 Inscribed Fields")]
    public float duration = 4; // Duration of interpolation movement

    private EnemyShield[] allShields;
    private EnemyShield thisShield;
    private Vector3 p0, p1; // The two points to interpolate
    private float timeStart; // Birth time for this Enemy_4

    void Start() {
        allShields = GetComponentsInChildren<EnemyShield>(); // a
        thisShield = GetComponent<EnemyShield>();

        // Initially set p0 & p1 to the current position (from Main.SpawnEnemy())
        p0 = p1 = pos; // a
        InitMovement();
    }

    void InitMovement() { // b
        p0 = p1; // Set p0 to the old p1
        // Assign a new on-screen location to p1
        float widMinRad = bndCheck.camWidth - bndCheck.radius;
        float hgtMinRad = bndCheck.camHeight - bndCheck.radius;
        p1.x = Random.Range( -widMinRad, widMinRad );
        p1.y = Random.Range( -hgtMinRad, hgtMinRad );
        // Make sure that it moves to a different quadrant of the screen
        if ( p0.x * p1.x > 0 && p0.y * p1.y > 0 ) { // c
            if ( Mathf.Abs( p0.x ) > Mathf.Abs( p0.y ) ) {
                p1.x *= -1;
            } else {
                p1.y *= -1;
            }
        }
        // Reset the time
        timeStart = Time.time;
    }

    public override void Move() {
        // This completely overrides Enemy.Move() with a linear interpolation
        float u = (Time.time -timeStart) / duration;
        if (u >= 1) {
            InitMovement();
            u = 0;
        }
        u = u - 0.15f * Mathf.Sin( u * 2 * Mathf.PI ); // Easing: Sine -0.15 // e
        pos = (1 - u) * p0 + u * p1; // Simple linear interpolation // f
    }

    /// <summary>
    /// Enemy_4 Collisions are handled differently from other Enemy subclasses
    /// to enable protection by EnemyShields.
    /// </summary>
    /// <param name="coll"></param>
    void OnCollisionEnter( Collision coll ) { // b
        GameObject otherGO = coll.gameObject;

        // Make sure this was hit by a ProjectileHero
        ProjectileHero p = otherGO.GetComponent<ProjectileHero>();
        if ( p != null ) {
            // Destroy the ProjectileHero regardless of bndCheck.isOnScreen
            Destroy( otherGO ); // c

            // Only damage this Enemy if it’s on screen
            if ( bndCheck.isOnScreen ) {
                // Find the GameObject of this Enemy_4 that was actually hit
                GameObject hitGO = coll.contacts[0].thisCollider.gameObject; // d
                if ( hitGO == otherGO ) { // e
                    hitGO = coll.contacts[0].otherCollider.gameObject;
                }

                // Get the damage amount from the Main WEAP_DICT.
                float dmg = Main.GET_WEAPON_DEFINITION( p.type ).damageOnHit;

                // Find the EnemyShield that was hit (if there was one)
                bool shieldFound = false;
                foreach ( EnemyShield es in allShields ) { // f
                    if ( es.gameObject == hitGO ) {
                        es.TakeDamage( dmg );
                        shieldFound = true;
                    }
                }
                if ( !shieldFound ) thisShield.TakeDamage( dmg ); // g

                // If thisShield is still active, then it has not been destroyed
                if ( thisShield.isActive ) return; // h

                // This ship was destroyed so tell Main about it
                if ( !calledShipDestroyed ) {
                    Main.SHIP_DESTROYED( this );
                    calledShipDestroyed = true;
                }

                // Destroy this Enemy_4
                Destroy( gameObject );
            }
        } else {
            Debug.Log( "Enemy_4 hit by non-ProjectileHero: " + otherGO.name );
        }
    }
}
```

```c# title="C# Parallax.cs"
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Parallax : MonoBehaviour {
    [Header("Inscribed")]
    public Transform playerTrans; // The player ship
    public Transform[] panels; // The scrolling foregrounds
    [Tooltip("Speed at which the panels move in Y")]
    public float scrollSpeed = -30f;
    [Tooltip("Controls how much panels react to player movement (Default 0.25)")]
    public float motionMult = 0.25f; // a

    private float panelHt; // Height of each panel
    private float depth; // Depth of panels (that is, pos.z)

    void Start () {
        panelHt = panels[0].localScale.y;
        depth = panels[0].position.z;

        // Set initial positions of panels
        panels[0].position = new Vector3(0,0,depth);
        panels[1].position = new Vector3(0,panelHt,depth);
    }

    void Update () {
        float tY, tX=0;
        tY= Time.time * scrollSpeed % panelHt + (panelHt *0.5f); // b

        if (playerTrans != null) {
            tX = -playerTrans.transform.position.x * motionMult; // c
            //tY += -poi.transform.position.y * motionMult;
        }

        // Position panels[0]
        panels[0].position = new Vector3(tX, tY, depth);
        // Position panels[1] where needed to make a continuous starfield // d
        if (tY >= 0) {
            panels[1].position = new Vector3(tX, tY -panelHt, depth);
        } else {
            panels[1].position = new Vector3(tX, tY +panelHt, depth);
        }
    }
}
```
